Ejemplo de comparación para el agua¶

En este ejemplo se obtienen los datos de equilibrio líquido-vapor y de vapor sobrecalentado para el agua, tanto mediante la búsqueda web en NIST como mediante el paquete CoolProp. Se estima el calor latente de vaporización a presión atmosférica y se realizan dos gráficos. El primero corresponde a la curva P-T del equilibrio líquido-vapor y el segundo es una fracción del gráfico T-h, incluyendo la zona de equilibrio líquido-vapor.

for Tini, P_iso in zip(T_sat_interp, P_iso_graf*1e5):
    T = np.linspace(Tini, Tmax_iso_graf, num_pointsT_iso_graf)
    P = np.ones((num_pointsT_iso_graf, 1))*P_iso
    h_cp = CP.PropsSI("Hmass", "P", P_iso, "T", T, s.english)/1000  # kJ/kg
    
    tabla_array = np.column_stack((T, P/1e5, h_cp))

    datos_vsc_cp.append(
        PropData(datos=tabla_array,
                 encabezados=encabezados,
                 unidades=unidades,
                 alias_base=ALIAS_BASE)
                 )

P_ref = 101325.0  # Pa
T_sat_ref = CP.PropsSI("T", "P", P_ref, "Q", 0, s.english)

h_fg = h_g - h_f      # Calor latente de vaporización

print("Calor latente de vaporización:")
print(
    (f" T_sat = {T_sat_ref:.2f} K "
     f"({T_sat_ref - 273.15:.2f} °C)")
     )
print(
    (f" h_f = {h_f/1000:.2f} kJ/kg\n"
     f" h_g = {h_g/1000:.2f} kJ/kg\n"
     f" h_fg = {h_fg/1000:.2f} kJ/kg")

import matplotlib.pyplot as plt

fig, ax = plt.subplots(figsize=(15, 10))

# Ordenar por temperatura para que las líneas y el relleno sigan la curva.
ax.plot(
    h_sat_web, T_sat_web, label="curva de saturación (NIST)", color="tab:blue"
    )

ax.fill_betweenx(
    y=tabla_liq["T"],
    x1=tabla_liq["h"],
    x2=tabla_vap["h"],
    color="tab:blue",
    alpha=0.12
    )

ax.plot(
    h_sat_cp, T_sat_cp,
    "--", label="curva de saturación (CoolProp)", color="tab:orange",
)

# Isobaras de vapor sobrecalentado (NIST)
for isobara in datos_vsc_web:
    ax.plot(
        isobara["h"],
        isobara["T"],
        linewidth=1,
        label=f"{isobara["P"][0]:g} bar (NIST)",
    )

# Isobaras de vapor sobrecalentado (CoolProp)

    )
